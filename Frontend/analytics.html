async function loadAnalytics(){ const response = await fetch("http://127.0.0.1:8000/transactions"); const data = await response.json(); document.getElementById( "totalTransactions" ).innerText = data.length; if(data.length === 0){ return; } let totalRisk = 0; let highestRisk = 0; let fraudCount = 0; data.forEach(t => { totalRisk += t.risk_score; if(t.risk_score > highestRisk){ highestRisk = t.risk_score; } if(t.risk_score >= 60){ fraudCount++; } }); let averageRisk = (totalRisk / data.length).toFixed(1); document.getElementById( "avgRisk" ).innerText = averageRisk + "%"; document.getElementById( "fraudCases" ).innerText = fraudCount; document.getElementById( "highestRisk" ).innerText = highestRisk + "%"; } loadAnalytics();
